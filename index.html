const app = initializeApp(firebaseConfig); const db = getFirestore(app); window.score = 0; let saveTimeout = null; // タイマー用の変数を追加 function updateUI(){ document.getElementById("score").innerText = window.score; } // スコア増加（画面はすぐ更新、保存は裏で遅延させる） window.addScore = function() { window.score += 1; updateUI(); // 前回の保存タイマーが残っていればキャンセル clearTimeout(saveTimeout); // 500ミリ秒（0.5秒）間、次のクリックがなければ保存を実行 saveTimeout = setTimeout(() => { saveScore(window.score); }, 500); }; // 保存 window.saveScore = async function(score){ try { await setDoc(doc(db, "scores", "player1"), { score: score }); console.log("スコアを保存しました:", score); } catch (error) { console.error("保存エラー:", error); } }; // 読み込み window.loadScore = async function(){ try { const docSnap = await getDoc(doc(db, "scores", "player1")); if (docSnap.exists()) { window.score = docSnap.data().score; } else { window.score = 0; } updateUI(); } catch (error) { console.error("読み込みエラー:", error); } }; window.addEventListener("load", loadScore);
